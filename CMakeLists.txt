cmake_minimum_required(VERSION 3.20)
project(wtfweg_shader_studio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL3 3.2 CONFIG REQUIRED)
find_path(VULKAN_INCLUDE_DIR vulkan/vulkan.h REQUIRED)
include(static_curl.cmake)
find_package(Threads REQUIRED)
include(source_files.cmake)
add_library(studio-model STATIC project.cpp shadertoy_project.cpp shadertoy_json.cpp ${STUDIO_MODEL_SHARED_SOURCES})
target_include_directories(studio-model PUBLIC "${SRC}/main" "${VULKAN_INCLUDE_DIR}")
target_include_directories(studio-model PRIVATE "${SRC}/deps/misc")
target_link_libraries(studio-model PUBLIC SDL3::SDL3 ${CMAKE_DL_LIBS})
add_library(studio-download STATIC shadertoy_download.cpp)
target_include_directories(studio-download PRIVATE "${SRC}/deps/misc")
target_link_libraries(studio-download PUBLIC studio-model studio-curl Threads::Threads)
add_executable(wtfweg-shader-studio main.cpp preview.cpp shadertoy_runtime.cpp shadertoy_smoke.cpp
  ${STUDIO_APP_SHARED_SOURCES})
target_include_directories(wtfweg-shader-studio PRIVATE
  "${SRC}/deps/imgui" "${SRC}/deps/misc" "${SRC}/deps/stb" "${SRC}/deps/ImGuiColorTextEdit")
target_link_libraries(wtfweg-shader-studio PRIVATE studio-model studio-download)
enable_testing()
add_test(NAME shader-studio-source-package COMMAND "${CMAKE_COMMAND}"
  "-DSTUDIO_PACKAGE_TEST_OUTPUT=${CMAKE_CURRENT_BINARY_DIR}/source-package-test-output"
  -P "${CMAKE_CURRENT_SOURCE_DIR}/package_source_smoke.cmake")
add_executable(shader-studio-download-smoke download_smoke.cpp)
target_link_libraries(shader-studio-download-smoke PRIVATE studio-download)
add_test(NAME shader-studio-download-smoke COMMAND shader-studio-download-smoke "${CMAKE_CURRENT_BINARY_DIR}/download-test-output")
if(CMAKE_OBJDUMP AND NOT APPLE)
  add_test(NAME shader-studio-static-curl COMMAND "${CMAKE_COMMAND}" "-DOBJDUMP=${CMAKE_OBJDUMP}"
    "-DBINARY=$<TARGET_FILE:wtfweg-shader-studio>" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_static_curl.cmake")
endif()
add_executable(shader-studio-project-smoke project_smoke.cpp)
target_link_libraries(shader-studio-project-smoke PRIVATE studio-model)
add_test(NAME shader-studio-project-smoke COMMAND shader-studio-project-smoke "${CMAKE_CURRENT_BINARY_DIR}/project-test-output")
option(STUDIO_GPU_TESTS "Register tests that require an OpenGL 4.5 context" OFF)
if(STUDIO_GPU_TESTS)
  add_test(NAME shader-studio-shadertoy-smoke COMMAND wtfweg-shader-studio --shadertoy-smoke-test "${CMAKE_CURRENT_BINARY_DIR}/shadertoy-test-output")
  set_tests_properties(shader-studio-shadertoy-smoke PROPERTIES TIMEOUT 60)
endif()
install(TARGETS wtfweg-shader-studio RUNTIME DESTINATION bin)
install(FILES README.md THIRD_PARTY_NOTICES.md "${STUDIO_SOURCE_ROOT}/LICENSE" DESTINATION share/wtfweg-shader-studio)
install(FILES "${SRC}/deps/ImGuiColorTextEdit/LICENSE" DESTINATION share/wtfweg-shader-studio/licenses/ImGuiColorTextEdit)
install(FILES "${SRC}/deps/librashader/UPSTREAM.txt" DESTINATION share/wtfweg-shader-studio/licenses/librashader)
install(FILES "${SRC}/deps/librashader/librashader.h" DESTINATION share/wtfweg-shader-studio/licenses/librashader)
install(FILES "${SRC}/deps/misc/cJSON.h" DESTINATION share/wtfweg-shader-studio/licenses/cJSON)
install(FILES CURL-LICENSE.txt DESTINATION share/wtfweg-shader-studio/licenses/curl)
install(FILES IMGUI-LICENSE.txt GLAD-LICENSE.txt DIRENT-LICENSE.txt
  DESTINATION share/wtfweg-shader-studio/licenses)
